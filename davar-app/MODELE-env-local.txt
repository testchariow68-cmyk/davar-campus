#  CE FICHIER EST PRIVÉ quand vous l'aurez rempli. Il ne va JAMAIS sur GitHub, et

#
#        davar-app\.env.local
#
#  Enregistré sous un autre nom, il ne sera PAS lu par l'application.

#  Les lignes commençant par # sont des explications : elles restent dans le

#
#  Règle d'écriture : NOM=valeur  — pas d'espace autour du =, pas de guillemets.
#
#  Plus simple encore : lancez « npm run env:local » — il écrit lui-même ce
#  fichier avec les trois clés déjà fabriquées, et ne remplace jamais une valeur
#  déjà présente.

# ── Le campus tourne en mode développement (sûr sur votre machine) ─────────────
APP_ENV=development


# ── 1. BASE DE DONNÉES (Turso) ────────────────────────────────────────────────
# Pour la VRAIE base Turso, collez son URL libsql://… après le =
# (sinon, laissez la base locale pour travailler hors ligne).
TURSO_DATABASE_URL=file:dev-data/davar-dev.db
# Le jeton d'accès de cette base (il commence par eyJ… et il est très long :
# c'est normal). À ne coller QUE si vous utilisez la vraie base Turso.
TURSO_AUTH_TOKEN=
# L'hôte seul, sans « libsql:// » ni barre oblique (pour le déploiement public).
# Exemple : mon-campus-davar-42.aws-eu-west-1.turso.io
TURSO_EXPECTED_HOST=


# ── 2. VOS CLÉS FABRIQUÉES (personne ne peut vous les donner) ─────────────────
# Ces trois valeurs se FABRIQUENT. Le plus simple : « npm run env:local ».
# Sinon, inventez-les : une longue chaîne au hasard, 32 caractères minimum.
# Attention : une fois le campus ouvert, ne les changez plus (cela déconnecterait
# les comptes déjà créés).
AUTH_PARAMS_SECRET=
AUTH_VERIFIER_PEPPER=
APP_DIAGNOSTIC_TOKEN=


# ── 3. ENVOI DES E-MAILS (Google Apps Script) ─────────────────────────────────
# « apps_script » est le bon mot : recopiez-le tel quel.
MAILER_KIND=apps_script
# L'URL du déploiement, celle qui finit par /exec
# Exemple : https://script.google.com/macros/s/AKfy…/exec
MAIL_APPS_SCRIPT_URL=
# La propriété DAVAR_MAIL_SECRET de votre script (32 caractères minimum)
MAIL_APPS_SCRIPT_TOKEN=


# ── 4. STOCKAGE DES FICHIERS (Cloudflare R2) ──────────────────────────────────
# ATTENTION : la clé d'accès et son secret viennent de « Gérer les clés d'API S3 »
# dans R2 — ce n'est PAS votre jeton d'API Cloudflare.
# R2_ACCOUNT_ID : 32 caractères hexadécimaux (l'identifiant de compte Cloudflare)
R2_ACCOUNT_ID=
R2_ACCESS_KEY_ID=
R2_SECRET_ACCESS_KEY=
# Le NOM de votre seau : minuscules, chiffres et tirets (pas de majuscules).
# ⚠️ Ce n'est PAS une valeur de jeton : c'est le nom affiché en haut de la page
# de votre seau, par exemple davar-campus-fichiers.
R2_BUCKET=


# ── 5. VENTE CHARIOW (automatisation Pulse) ───────────────────────────────────
# Le drapeau reste FERMÉ jusqu'à la recette sur le compte marchand : ne le
# passez pas à true maintenant.
CHARIOW_ENABLE_PULSE=false
# Le secret de signature du Pulse (whsec_…) — visible dans l'onglet « Aperçu »
# du Pulse, bloc « Secret de signature » → Révéler. Ce n'est PAS votre clé API.
CHARIOW_PULSE_SECRET=
# L'identifiant du Pulse (pulse_…) et celui de la boutique (str_…)
CHARIOW_PULSE_ID=
CHARIOW_STORE_ID=
# La clé API de la boutique (sk_…)
CHARIOW_API_KEY=


# ── 6. ASSISTANTS (la même clé sert la dictée vocale des avis) ─────────────────
# Le palier gratuit de Groq est le plus généreux : commencez par celui-là.
# Sa clé commence par gsk_… ; sans elle, l'assistant le dit honnêtement et
# l'étudiant qui dicte un avis bascule sur son appareil.
GROQ_API_KEY=
# Facultatif : second moteur, si vous voulez une réserve (clé AIza…)
GEMINI_API_KEY=
